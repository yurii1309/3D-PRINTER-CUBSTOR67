        categoryName: catObj.name,

        inStock: productForm.inStock

      description: prod.description,
      image: prod.image,
      colors: Array.isArray(prod.colors) ? prod.colors.join(', ') : prod.colors,
      material: prod.material || 'PLA Plastic',

                        <button
                          onClick={() => handleDeleteProduct(product.id)}
                          className="p-2 bg-rose-600 hover:bg-rose-500 text-white rounded-xl shadow-lg font-bold transition-all"
                          title="Видалити"
                        >
                          <Trash2 className="w-3.5 h-3.5" />
                        </button>
                      </div>
                    )}
                  </div>

                  <div className="p-5 flex-1 flex flex-col justify-between space-y-4">
                    <div className="space-y-1.5">
                      <span className="text-[10px] text-cyan-400 font-bold uppercase tracking-wider">{product.categoryName}</span>
                      <h3 className="text-base font-bold text-white line-clamp-1 group-hover:text-cyan-300 transition-colors">{product.title}</h3>
                      <p className="text-xs text-slate-400 line-clamp-2 leading-relaxed">{product.description}</p>
                    </div>

                    {product.colors && (
                      <div className="text-[11px] text-slate-400">
                        <span className="text-slate-500 font-medium">Кольори:</span> {Array.isArray(product.colors) ? product.colors.join(', ') : product.colors}
                      </div>
                    )}

                    <div className="pt-3 border-t border-slate-800/80 flex items-center justify-between gap-2">
                      <div>
                        <span className="text-lg font-black text-white">{product.price} ₴</span>
                        {product.oldPrice && (
                          <span className="text-xs text-slate-500 line-through block -mt-1">{product.oldPrice} ₴</span>
                        )}
                      </div>

                      <button
                        disabled={!product.inStock}
                        onClick={() => addToCart(product)}
                        className={`px-4 py-2.5 rounded-xl font-bold text-xs flex items-center gap-1.5 transition-all ${
                          product.inStock 
                            ? 'bg-cyan-500 hover:bg-cyan-400 text-slate-950 shadow-lg shadow-cyan-500/20 active:scale-95' 
                            : 'bg-slate-800 text-slate-500 cursor-not-allowed'
                        }`}
                      >
                        <ShoppingBag className="w-3.5 h-3.5" />
                        Замовити
                      </button>
                    </div>
                  </div>
                </div>
              ))}
            </div>
          </>
        )}

      </main>

      {/* 4. МОДАЛЬНЕ ВІКНО ВХОДУ (КОД: 1309) */}
      {showAdminModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-950/80 backdrop-blur-md">
          <div className="bg-slate-900 border border-purple-500/30 rounded-3xl p-6 max-w-xs w-full space-y-4 shadow-2xl relative">
            <div className="flex justify-between items-center border-b border-slate-800 pb-3">
              <h3 className="text-sm font-bold text-white flex items-center gap-2">
                <Lock className="w-4 h-4 text-purple-400" /> Вхід Продавця
              </h3>
              <button onClick={() => setShowAdminModal(false)} className="text-slate-400 hover:text-white">
                <X className="w-4 h-4" />
              </button>
            </div>

            <form onSubmit={handleAdminLogin} className="space-y-3">
              <div>
                <label className="text-xs text-slate-300 font-semibold block mb-1">PIN-код продавця:</label>
                <input
                  type="password"
                  placeholder="1309"
                  value={pinCode}
                  onChange={(e) => { setPinCode(e.target.value); setPinError(''); }}
                  className="w-full bg-slate-950 border border-slate-800 focus:border-purple-500 rounded-xl px-3 py-2.5 text-center text-lg font-mono text-white focus:outline-none"
                  maxLength={6}
                  autoFocus
                />
                {pinError && <p className="text-[11px] text-rose-400 mt-1.5 text-center">{pinError}</p>}
              </div>

              <button type="submit" className="w-full bg-purple-600 hover:bg-purple-500 text-white font-bold py-2.5 rounded-xl text-xs transition-all">
                Увійти
              </button>
            </form>
          </div>
        </div>
      )}

      {/* 5. МОДАЛЬНЕ ВІКНО СТВОРЕННЯ / РЕДАГУВАННЯ */}
      {showProductModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-950/80 backdrop-blur-md">
          <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 max-w-md w-full space-y-4 max-h-[90vh] overflow-y-auto shadow-2xl">
            <div className="flex justify-between items-center border-b border-slate-800 pb-3">
              <h3 className="text-base font-bold text-white">{editingProduct ? 'Редагувати' : 'Нове оголошення'}</h3>
              <button onClick={() => setShowProductModal(false)} className="text-slate-400 hover:text-white">
                <X className="w-4 h-4" />
              </button>
            </div>

            <form onSubmit={handleSaveProduct} className="space-y-3.5 text-xs">
              <div>
                <label className="text-slate-300 font-semibold block mb-1">Назва товару</label>
                <input required type="text" value={productForm.title} onChange={(e) => setProductForm({ ...productForm, title: e.target.value })} className="w-full bg-slate-950 border border-slate-800 focus:border-cyan-500 rounded-xl px-3 py-2 text-white focus:outline-none" />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="text-slate-300 font-semibold block mb-1">Категорія</label>
                  <select value={productForm.category} onChange={(e) => setProductForm({ ...productForm, category: e.target.value })} className="w-full bg-slate-950 border border-slate-800 focus:border-cyan-500 rounded-xl px-3 py-2 text-white focus:outline-none">
                    {CATEGORIES.filter(c => c.id !== 'all').map(c => <option key={c.id} value={c.id}>{c.name}</option>)}
                  </select>
                </div>
                <div>
                  <label className="text-slate-300 font-semibold block mb-1">Ціна (₴)</label>
                  <input required type="number" value={productForm.price} onChange={(e) => setProductForm({ ...productForm, price: e.target.value })} className="w-full bg-slate-950 border border-slate-800 focus:border-cyan-500 rounded-xl px-3 py-2 text-white focus:outline-none" />
                </div>
              </div>

              <div>
                <label className="text-slate-300 font-semibold block mb-1">URL фотографії</label>
                <input type="text" value={productForm.image} onChange={(e) => setProductForm({ ...productForm, image: e.target.value })} className="w-full bg-slate-950 border border-slate-800 focus:border-cyan-500 rounded-xl px-3 py-2 text-white focus:outline-none" />
              </div>

              <div>
                <label className="text-slate-300 font-semibold block mb-1">Опис</label>
                <textarea rows={3} value={productForm.description} onChange={(e) => setProductForm({ ...productForm, description: e.target.value })} className="w-full bg-slate-950 border border-slate-800 focus:border-cyan-500 rounded-xl px-3 py-2 text-white focus:outline-none" />
              </div>

              <button type="submit" className="w-full bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-black py-3 rounded-xl shadow-lg shadow-cyan-500/20 text-xs transition-all">
                Зберегти оголошення
              </button>
            </form>
          </div>
        </div>
      )}

      {/* 6. КОШИК */}
      {isCartOpen && (
        <div className="fixed inset-0 z-50 flex justify-end bg-slate-950/80 backdrop-blur-md">
          <div className="w-full max-w-md bg-slate-900 h-full shadow-2xl flex flex-col border-l border-slate-800">
            <div className="p-5 border-b border-slate-800 flex justify-between items-center">
              <h3 className="font-bold text-white flex items-center gap-2 text-sm"><ShoppingBag className="w-4 h-4 text-cyan-400" /> Ваші замовлення ({cart.length})</h3>
              <button onClick={() => setIsCartOpen(false)} className="text-slate-400 hover:text-white"><X className="w-5 h-5" /></button>
            </div>

            <div className="flex-1 overflow-y-auto p-5 space-y-3">
              {cart.length === 0 ? (
                <div className="text-center py-16 text-slate-500 text-xs">Кошик порожній</div>
              ) : (
                cart.map(item => (
                  <div key={item.id} className="flex items-center gap-3 p-3 bg-slate-950 rounded-2xl border border-slate-800">
                    <img src={item.image} alt={item.title} className="w-12 h-12 rounded-xl object-cover" />
                    <div className="flex-1 min-w-0">
                      <h4 className="text-xs font-bold text-white truncate">{item.title}</h4>
                      <span className="text-xs font-bold text-cyan-400">{item.price * item.quantity} ₴</span>
                    </div>
                    <div className="flex items-center gap-2">
                      <button onClick={() => setCart(prev => prev.map(i => i.id === item.id ? { ...i, quantity: Math.max(0, i.quantity - 1) } : i).filter(i => i.quantity > 0))} className="p-1 text-slate-400 hover:text-white"><Minus className="w-3.5 h-3.5" /></button>
                      <span className="text-xs font-bold text-white">{item.quantity}</span>
                      <button onClick={() => setCart(prev => prev.map(i => i.id === item.id ? { ...i, quantity: i.quantity + 1 } : i))} className="p-1 text-slate-400 hover:text-white"><Plus className="w-3.5 h-3.5" /></button>
                    </div>
                  </div>
                ))
              )}
            </div>

            {cart.length > 0 && (
              <div className="p-5 border-t border-slate-800 space-y-3 bg-slate-950">
                <div className="flex justify-between text-sm font-bold text-white">
                  <span>Загальна сума:</span>
                  <span className="text-cyan-400">{cartTotal} ₴</span>
                </div>
                <button onClick={() => { setIsCartOpen(false); setIsCheckoutOpen(true); }} className="w-full bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-black py-3 rounded-xl text-xs shadow-lg shadow-cyan-500/20">
                  Оформити замовлення
                </button>
              </div>
            )}
          </div>
        </div>
      )}

      {/* 7. ОФОРМЛЕННЯ */}
      {isCheckoutOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-950/80 backdrop-blur-md">
          <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 max-w-sm w-full space-y-4 shadow-2xl">
            <div className="flex justify-between items-center border-b border-slate-800 pb-3">
              <h3 className="font-bold text-white text-sm">Доставка Новою Поштою</h3>
              <button onClick={() => setIsCheckoutOpen(false)} className="text-slate-400 hover:text-white"><X className="w-4 h-4" /></button>
            </div>

            {orderSuccess ? (
              <div className="text-center py-6 space-y-3">
                <CheckCircle2 className="w-12 h-12 text-emerald-400 mx-auto" />
                <h4 className="font-bold text-white">Замовлення прийнято!</h4>
                <p className="text-xs text-slate-400">Продавець зв'яжеться з вами для підтвердження.</p>
                <button onClick={() => { setCart([]); setIsCheckoutOpen(false); setOrderSuccess(false); }} className="bg-cyan-500 text-slate-950 font-black text-xs px-5 py-2.5 rounded-xl">
                  Зрозуміло
                </button>
              </div>
            ) : (
              <form onSubmit={(e) => { e.preventDefault(); setOrderSuccess(true); }} className="space-y-3 text-xs">
                <div>
                  <label className="text-slate-300 font-semibold block mb-1">ПІБ</label>
                  <input required type="text" placeholder="Іван Іваненко" className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-white focus:outline-none" />
                </div>
                <div>
                  <label className="text-slate-300 font-semibold block mb-1">Телефон</label>
                  <input required type="tel" placeholder="+380..." className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-white focus:outline-none" />
                </div>
                <div>
                  <label className="text-slate-300 font-semibold block mb-1">Місто та Номер відділення НП</label>
                  <input required type="text" placeholder="Київ, №15" className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-white focus:outline-none" />
                </div>

                <div className="p-3 bg-slate-950 rounded-xl flex justify-between font-bold text-white text-xs border border-slate-800">
                  <span>До сплати:</span>
                  <span className="text-cyan-400">{cartTotal} ₴</span>
                </div>

                <button type="submit" className="w-full bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-black py-3 rounded-xl shadow-lg shadow-cyan-500/20">
                  Підтвердити замовлення
                </button>
              </form>
            )}
          </div>
        </div>
      )}

    </div>
  );
}
